\weq{9}{\wp(nu) = \wp(u) - \frac{\psi_{n+1}(u)\,\psi_{n-1}(u)}{\psi_n(u)^2},}

\weq{10}{
\begin{array}{ll}
\wp(nu) = \wp(u) - \dfrac{P_{n+1}\,P_{n-1}}{\wp'(u)^2\,P_n^2}, & n\text{ even,}\\[3mm]
\phantom{\wp(nu)}{}= \wp(u) - \dfrac{\wp'(u)^2\,P_{n+1}\,P_{n-1}}{P_n^2}, & n\text{ odd.}
\end{array}}

We determine first of all the function $P_n$ in the first cases $n=1, 2, 3,
4$. For this we form, according to the addition theorem of the $\wp$-function
[§ 49, (11)]:
\[
4\big[\wp(u) + \wp(v) + \wp(u+v)\big] = \left(\frac{\wp'(u)-\wp'(v)}{\wp(u)-\wp(v)}\right)^2,
\]
by setting $u=v$ and determining the limiting value on the right by
differentiation:
\[
\wp(2u) + 2\wp(u) = \frac14\left(\frac{\wp''(u)}{\wp'(u)}\right)^2
= \frac{\frac14\left(6\wp(u)^2 - \frac12 g_2\right)^2}{4\wp(u)^3 - g_2\wp(u) - g_3},
\]

\weq{11}{\wp(2u) = \wp(u) - \frac{3\wp(u)^4 - \frac32 g_2\wp(u)^2 - 3g_3\wp(u) - \frac{g_2^2}{16}}{\wp'(u)^2},}

\[
\psi_1 = 1,\quad \psi_2 = -\wp'(u),\quad \psi_3 = P_3,
