\weq{5}{\begin{aligned} U &= (a_1, m_1 b, m_1^2 a_2 c), \\ V &= (a_2, m_2 b, m_2^2 a_1 c), \end{aligned}}

\weq{6}{T = (a_1 a_2, b, c).}
