\gesperrt{Table V.}\\
\gesperrt{Development of the transcendents of the second kind} (p.\ 164).
\end{center}

\weq{1}{\begin{aligned}
\frac{d\log H(2Ku)}{\pi\,du} &= \frac{2K}{\pi} Z(2Ku) + \frac{d\log \operatorname{sn}2Ku}{\pi\,du} \\
&= \operatorname{cotg}\pi u + 4\sin2\pi u \sum \frac{q^m}{1-2q^m\cos2\pi u+q^{2m}}, \\
&= \operatorname{cotg}\pi u + 4\sum \frac{q^m}{1-q^m}\sin m\pi u.
\end{aligned}}

\weq{2}{\begin{aligned}
\frac{d\log H(2Ku+K)}{\pi\,du} &= \frac{2K}{\pi} Z(2Ku) + \frac{d\log \operatorname{cn}2Ku}{\pi\,du}, \\
&= -\operatorname{tg}\pi u - 4\sin2\pi u \sum \frac{q^m}{1+2q^m\cos2\pi u+q^m}, \\
&= -\operatorname{tg}\pi u + 4\sum \frac{(-1)^{\frac{m}{2}} q^m}{1-q^m}\sin m\pi u.
\end{aligned}} % [sic] denominator's last term printed q^m, not q^{2m} as in the neighbouring formulas

\weq{3}{\begin{aligned}
\frac{d\log \Theta(2Ku)}{\pi\,du} &= \frac{2K}{\pi} Z(2Ku), \\
&= 4\sin2\pi u \sum \frac{q^n}{1-2q^n\cos2\pi u+q^{2n}}, \\
&= 4\sum \frac{q^{\frac{m}{2}}\sin\pi m u}{1-q^m}.
\end{aligned}}

\weq{4}{\begin{aligned}
\frac{d\log \Theta(2Ku+K)}{\pi\,du} &= \frac{2K}{\pi} Z(2Ku) + \frac{d\log \operatorname{dn}2Ku}{\pi\,du}, \\
&= -4\sin2\pi u \sum \frac{q^n}{1+2q^n\cos2\pi u+q^{2n}}, \\
&= 4\sum (-1)^{\frac{m}{2}} \frac{q^{\frac{m}{2}}}{1-q^m}\sin m\pi u.
\end{aligned}}
