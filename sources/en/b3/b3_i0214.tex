\weq{5}{
\begin{array}{ll}
\left(\dfrac{z}{\sqrt{\varkappa'}}\right)^{n^2-1}\! A\!\left(\dfrac{y^2}{z^2}\right) = (-1)^{\frac{n-1}{2}} B(x^2), & A(1) = (-1)^{\frac{n-1}{2}}\, \sqrt{\varkappa'}^{\,n^2-1}, \\[8pt]
\left(\dfrac{z}{\sqrt{\varkappa'}}\right)^{n^2-1}\! B\!\left(\dfrac{y^2}{z^2}\right) = (-1)^{\frac{n-1}{2}} A(x^2), & B(1) = (-1)^{\frac{n-1}{2}}\, n\, \sqrt{\varkappa'}^{\,n^2-1}, \\[8pt]
\left(\dfrac{z}{\sqrt{\varkappa'}}\right)^{n^2-1}\! C\!\left(\dfrac{y^2}{z^2}\right) = D(x^2), & C(1) = \sqrt{\varkappa'}^{\,n^2-1}, \\[8pt]
\left(\dfrac{z}{\sqrt{\varkappa'}}\right)^{n^2-1}\! D\!\left(\dfrac{y^2}{z^2}\right) = C(x^2), & D(1) = \sqrt{\varkappa'}^{\,n^2-1}, \\[8pt]
& n \text{ odd.}
\end{array}
}
\weq{6}{
\begin{array}{ll}
(\sqrt{\varkappa}\,x)^{n^2-4}\, A\!\left(\dfrac{1}{\varkappa^2 x^2}\right) = (-1)^{\frac{n}{2}-1} A(x^2), & A(\infty) = (-1)^{\frac{n}{2}-1}\, n\, (\sqrt{\varkappa})^{n^2-4}, \\[8pt] % [sic] German source prints a positive exponent; the parallel rows suggest -(n^2-4).
(\sqrt{\varkappa}\,x)^{n^2}\, B\!\left(\dfrac{1}{\varkappa^2 x^2}\right) = B(x^2), & B(\infty) = \sqrt{\varkappa}^{\,-n^2}, \\[8pt]
(\sqrt{\varkappa}\,x)^{n^2}\, C\!\left(\dfrac{1}{\varkappa^2 x^2}\right) = C(x^2), & C(\infty) = \sqrt{\varkappa}^{\,-n^2}, \\[8pt]
(\sqrt{\varkappa}\,x)^{n^2}\, D\!\left(\dfrac{1}{\varkappa^2 x^2}\right) = (-1)^{\frac{n}{2}} D(x^2), & D(\infty) = (-1)^{\frac{n}{2}}\, \sqrt{\varkappa}^{\,-n^2}, \\[8pt]
& n \text{ even.}
\end{array}
}
\weq{7}{
\begin{array}{ll}
(\sqrt{\varkappa}\,x)^{n^2-1}\, A\!\left(\dfrac{1}{\varkappa^2 x^2}\right) = (-1)^{\frac{n-1}{2}} D(x^2), & A(\infty) = (-1)^{\frac{n-1}{2}}\, \sqrt{\varkappa}^{\,-(n^2-1)}, \\[8pt]
(\sqrt{\varkappa}\,x)^{n^2-1}\, B\!\left(\dfrac{1}{\varkappa^2 x^2}\right) = C(x^2), & B(\infty) = \sqrt{\varkappa}^{\,-(n^2-1)}, \\[8pt]
(\sqrt{\varkappa}\,x)^{n^2-1}\, C\!\left(\dfrac{1}{\varkappa^2 x^2}\right) = B(x^2), & C(\infty) = \sqrt{\varkappa}^{\,-(n^2-1)}, \\[8pt]
(\sqrt{\varkappa}\,x)^{n^2-1}\, D\!\left(\dfrac{1}{\varkappa^2 x^2}\right) = (-1)^{\frac{n-1}{2}} A(x^2), & D(\infty) = (-1)^{\frac{n-1}{2}}\, n\, \sqrt{\varkappa}^{\,n^2-1}, \\[8pt] % [sic] German source prints a positive exponent; the parallel rows suggest -(n^2-1).
& n \text{ odd.}
\end{array}
}

Here $A(\infty)$, $B(\infty)$, $C(\infty)$, $D(\infty)$ are in every case to be
understood as the coefficients of the highest power of $x$ in these functions. In
(7) the last two formulae can be derived from the first by the interchange of $x$
with $1/\varkappa x$.

The third system of formulae can be obtained either in the same way, or else
by replacing $x$ in the formulae (4), (5) by $1/\varkappa x$ and applying
(6), (7).
