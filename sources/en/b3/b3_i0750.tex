\weq{7}{\begin{aligned}
\frac{2\varkappa' K}{\pi}\frac{1}{\operatorname{cn}2Ku} &= \frac{1}{\cos\pi u} - 4\cos\pi u \sum \frac{(-1)^{\frac{m}{2}} q^{\frac{n}{2}}(1+q^m)}{1+2q^m\cos2\pi u+q^{2m}} \\
&= \frac{1}{\cos\pi u} - 4\sum (-1)^{\frac{n-1}{2}} \frac{q^n}{1+q^n}\cos n\pi u.
\end{aligned}} % [sic] q exponent printed n/2 rather than m/2 (the sum runs over even m); also printed -4\cos\pi u before the first sum, wrong sign: the second line expands to +4\cos\pi u (at u=0 the first line, read with q^{m/2}, gives 1+4q+..., while 2\varkappa'K/\pi = 1-4q+...)

\weq{8}{\begin{aligned}
\frac{2\varkappa' K}{\pi}\frac{\operatorname{sn}Ku}{\operatorname{cn}2Ku} &= \operatorname{tg}\pi u + 4\sin2\pi u \sum \frac{(-1)^{\frac{m}{2}} q^m}{1-q^m\cos2\pi u+q^{2m}} \\
&= \operatorname{tg}\pi u + 4\sum (-1)^{\frac{m}{2}} \frac{q^m}{1+q^m}\sin m\pi u.
\end{aligned}} % [sic] printed sn Ku (without 2) and denominator omits the coefficient 2; the printed denominator 1-q^m\cos2\pi u+q^{2m} also has the wrong sign, 1+2q^m\cos2\pi u+q^{2m} is required as in (7) and (9)

\weq{9}{\begin{aligned}
\frac{2K}{\pi}\frac{\operatorname{dn}2Ku}{\operatorname{cn}2Ku} &= \frac{1}{\cos\pi u} + 4\cos\pi u \sum \frac{q^{\frac{m}{2}}(1+q^m)}{1+2q^m\cos2\pi u+q^{2m}} \\
&= \frac{1}{\cos\pi u} + 4\sum (-1)^{\frac{n-1}{2}} \frac{q^n}{1-q^n}\cos n\pi u.
\end{aligned}}

\weq{10}{\begin{aligned}
\frac{2\varkappa' K}{\pi}\frac{1}{\operatorname{dn}2Ku} &= 1-4\sum (-1)^{\frac{n-1}{2}} \frac{q^n(q^n+\cos2\pi u)}{1+2q^n\cos2\pi u+q^{2n}} \\
&= 1+4\sum (-1)^{\frac{m}{2}} \frac{q^{\frac{m}{2}}}{1+q^m}\cos m\pi u.
\end{aligned}}

\weq{11}{\begin{aligned}
\frac{2\varkappa\varkappa' K}{\pi}\frac{\operatorname{sn}2Ku}{\operatorname{dn}2Ku} &= 4\sin\pi u \sum \frac{(-1)^{\frac{n-1}{2}} q^{\frac{n}{2}}(1-q^n)}{1+2q^n\cos2\pi u+q^{2n}} \\
&= 4\sum \frac{q^{\frac{n}{2}}}{1-q^n}\sin n\pi u.
% [sic] printed 4\sum q^{n/2}/(1-q^n)\sin n\pi u, which is the series of (2\varkappa K/\pi) sn 2Ku; the left side and the first line give 4\sum(-1)^{(n-1)/2} q^{n/2}/(1+q^n)\sin n\pi u
\end{aligned}}

\weq{12}{\begin{aligned}
