\weq{13}{
\begin{aligned}[t]
\frac{\vartheta'_{11}\vartheta_{00}(v+a)}{\pi\vartheta_{00}(v)\vartheta_{11}(a)} &= \frac{1}{\sin \pi a} \\
&\quad {} + 2i\sum\left[\frac{q^n e^{n\pi i a}}{e^{-2\pi i v}+q^n}-\frac{q^n e^{-n\pi i a}}{e^{2\pi i v}+q^n}\right].
\end{aligned}
}

\weq{14}{
\begin{aligned}[t]
\frac{\vartheta'_{11}\vartheta_{01}(v+a)}{\pi\vartheta_{00}(v)\vartheta_{10}(a)} &= \frac{1}{\cos \pi a} \\
&\quad {} - 2\sum(-1)^{\frac{n-1}{2}}\left[\frac{q^n e^{n\pi i a}}{e^{-2\pi i v}+q^n}+\frac{q^n e^{-n\pi i a}}{e^{2\pi i v}+q^n}\right].
\end{aligned}
}

\weq{15}{
\frac{\vartheta'_{11}\vartheta_{10}(v+a)}{\pi\vartheta_{00}(v)\vartheta_{01}(a)}
= 2\sum\left[\frac{q^{\frac{n}{2}} e^{n\pi i a} e^{-\pi i v}}{e^{-2\pi i v}+q^n}+\frac{q^{\frac{n}{2}} e^{-n\pi i a} e^{\pi i v}}{e^{2\pi i v}+q^n}\right].
}

\weq{16}{
\frac{\vartheta'_{11}\vartheta_{11}(v+a)}{\pi\vartheta_{00}(v)\vartheta_{00}(a)}
= -2i\sum(-1)^{\frac{n-1}{2}}\left[\frac{q^{\frac{n}{2}} e^{n\pi i a} e^{-\pi i v}}{e^{-2\pi i v}+q^n}-\frac{q^{\frac{n}{2}} e^{-n\pi i a} e^{\pi i v}}{e^{2\pi i v}+q^n}\right].
}
