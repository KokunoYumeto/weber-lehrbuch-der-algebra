\begin{center}
\gesperrt{Table} II.

\gesperrt{Second form of the development of the sixteen
$\vartheta$-quotients} (p. 91).
\end{center}

In Tables II, III, IV, V, $m$ runs through the sequence of the
positive even numbers, $n$ through the sequence of the positive odd
numbers, thus:
\[
\begin{aligned}
m &= 2, 4, 6, 8, 10, \ldots, \\
n &= 1, 3, 5, 7, 9, \ldots
\end{aligned}
\]

Likewise $m'$ is to run through the even numbers, $n'$ through the odd numbers.

\weq{1}{
\begin{aligned}[t]
\frac{\vartheta'_{11}\vartheta_{11}(v+a)}{\pi\vartheta_{11}(v)\vartheta_{11}(a)} &= \cotg \pi v + \cotg \pi a \\
&\quad {} - 2i\sum\left[\frac{q^m e^{m\pi i a}}{e^{-2\pi i v}-q^m}-\frac{q^m e^{-m\pi i a}}{e^{2\pi i v}-q^m}\right].
\end{aligned}
}

\weq{2}{
\begin{aligned}[t]
\frac{\vartheta'_{11}\vartheta_{10}(v+a)}{\pi\vartheta_{11}(v)\vartheta_{10}(a)} &= \cotg \pi v - \tg \pi a \\
&\quad {} - 2i\sum(-1)^{\frac{m}{2}}\left[\frac{q^m e^{m\pi i a}}{e^{-2\pi i v}-q^m}-\frac{q^m e^{-m\pi i a}}{e^{2\pi i v}-q^m}\right].
\end{aligned}
}

\weq{3}{
\begin{aligned}[t]
\frac{\vartheta'_{11}\vartheta_{01}(v+a)}{\pi\vartheta_{11}(v)\vartheta_{01}(a)} &= \frac{1}{\sin \pi v} \\
&\quad {} + 2i\sum\left[\frac{q^{\frac{m}{2}} e^{-m\pi i a} e^{\pi i v}}{e^{2\pi i v}-q^m}-\frac{q^{\frac{m}{2}} e^{m\pi i a} e^{-\pi i v}}{e^{-2\pi i v}-q^m}\right].
\end{aligned}
}

\weq{4}{
\begin{aligned}[t]
\frac{\vartheta'_{11}\vartheta_{00}(v+a)}{\pi\vartheta_{11}(v)\vartheta_{00}(a)} &= \frac{1}{\sin \pi v} \\
&\quad {} + 2i\sum(-1)^{\frac{m}{2}}\left[\frac{q^{\frac{m}{2}} e^{-m\pi i a} e^{\pi i v}}{e^{2\pi i v}-q^m}-\frac{q^{\frac{m}{2}} e^{m\pi i a} e^{-\pi i v}}{e^{-2\pi i v}-q^m}\right].
\end{aligned}
}
