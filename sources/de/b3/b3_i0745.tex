\weq{6}{\frac{\vartheta'_{11}\,\vartheta_{11}(v+a)}{\pi\,\vartheta_{10}(v)\,\vartheta_{10}(a)} = -\,\mathrm{tg}\,\pi\,v + \mathrm{tg}\,\pi\,a + 2i\sum(-1)^{\frac{m}{2}}\left[\frac{q^m\,e^{-m\pi i a}}{e^{2\pi i v}+q^m} - \frac{q^m\,e^{m\pi i a}}{e^{-2\pi i v}+q^m}\right].}
