\begin{center}
\gesperrt{Tabelle I.}

\gesperrt{Entwickelungen der sechzehn $\vartheta$-Quotienten} (S.\,88).

$\nu$ durchläuft alle ganzen Zahlen von $-\infty$ bis $+\infty$.
\end{center}

\weq{1}{\frac{\vartheta'_{11}\,\vartheta_{11}(v+a)}{\pi\,\vartheta_{11}(v)\,\vartheta_{11}(a)} = 2i\sum\frac{e^{2\pi i a \nu}}{q^{2\nu}e^{2\pi i v}-1}.}

\weq{2}{\frac{\vartheta'_{11}\,\vartheta_{10}(v+a)}{\pi\,\vartheta_{11}(v)\,\vartheta_{10}(a)} = 2i\sum\frac{(-1)^{\nu}e^{2\pi i a \nu}}{q^{2\nu}e^{2\pi i v}-1}.}

\weq{3}{\frac{\vartheta'_{11}\,\vartheta_{01}(v+a)}{\pi\,\vartheta_{11}(v)\,\vartheta_{01}(a)} = 2i\sum\frac{q^{\nu}\,e^{\pi i v}\,e^{2\pi i a \nu}}{q^{2\nu}e^{2\pi i v}-1}.}

\weq{4}{\frac{\vartheta'_{11}\,\vartheta_{00}(v+a)}{\pi\,\vartheta_{11}(v)\,\vartheta_{00}(a)} = 2i\sum\frac{(-1)^{\nu}q^{\nu}\,e^{\pi i v}\,e^{2\pi i a \nu}}{q^{2\nu}e^{2\pi i v}-1}.}

\medskip
\weq{5}{\frac{\vartheta'_{11}\,\vartheta_{10}(v+a)}{\pi\,\vartheta_{10}(v)\,\vartheta_{11}(a)} = -2i\sum\frac{e^{2\pi i a \nu}}{q^{2\nu}e^{2\pi i v}+1}.}

\weq{6}{\frac{\vartheta'_{11}\,\vartheta_{11}(v+a)}{\pi\,\vartheta_{10}(v)\,\vartheta_{10}(a)} = 2i\sum\frac{(-1)^{\nu}e^{2\pi i a \nu}}{q^{2\nu}e^{2\pi i v}+1}.}

\weq{7}{\frac{\vartheta'_{11}\,\vartheta_{00}(v+a)}{\pi\,\vartheta_{10}(v)\,\vartheta_{01}(a)} = 2\sum\frac{e^{2\pi i a \nu}\,q^{\nu}\,e^{\pi i v}}{q^{2\nu}e^{2\pi i v}+1}.}

\weq{8}{\frac{\vartheta'_{11}\,\vartheta_{01}(v+a)}{\pi\,\vartheta_{10}(v)\,\vartheta_{00}(a)} = 2\sum(-1)^{\nu}\frac{e^{2\pi i a \nu}\,q^{\nu}\,e^{\pi i v}}{q^{2\nu}e^{2\pi i v}+1}.}

\medskip
\weq{9}{\frac{\vartheta'_{11}\,\vartheta_{01}(v+a)}{\pi\,\vartheta_{01}(v)\,\vartheta_{11}(a)} = 2i\sum\frac{e^{(2\nu+1)\pi i a}}{q^{2\nu+1}e^{2\pi i v}-1}.}

\weq{10}{\frac{\vartheta'_{11}\,\vartheta_{00}(v+a)}{\pi\,\vartheta_{01}(v)\,\vartheta_{10}(a)} = -2\sum\frac{(-1)^{\nu}e^{(2\nu+1)\pi i a}}{q^{2\nu+1}e^{2\pi i v}-1}.}

\weq{11}{\frac{\vartheta_{11}\,\vartheta_{11}(v+a)}{\pi\,\vartheta_{01}(v)\,\vartheta_{01}(a)} = 2i\sum\frac{e^{(2\nu+1)\pi i a}\,q^{\frac{2\nu+1}{2}}\,e^{\pi i v}}{q^{2\nu+1}e^{2\pi i v}-1}.} % [sic] printed without a prime on the first vartheta, unlike the other fifteen lines

\weq{12}{\frac{\vartheta'_{11}\,\vartheta_{10}(v+a)}{\pi\,\vartheta_{01}(v)\,\vartheta_{00}(a)} = -2\sum\frac{(-1)^{\nu}e^{(2\nu+1)\pi i a}\,q^{\frac{2\nu+1}{2}}\,e^{\pi i v}}{q^{2\nu+1}e^{2\pi i v}-1}.}
