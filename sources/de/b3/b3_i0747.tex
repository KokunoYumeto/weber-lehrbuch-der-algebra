\begin{center}
\gesperrt{Tabelle} III.

\gesperrt{Entwickelung der $\vartheta$-Quotienten in trigonometrischen
Reihen} (S. 92).
\end{center}

\weq{1}{
\frac{\vartheta'_{11}\vartheta_{11}(v+a)}{\pi\vartheta_{11}(v)\vartheta_{11}(a)}
= \cotg \pi v + \cotg \pi a + 4\sum q^{\frac{mm'}{2}}\sin \pi (m a + m' v).
}

\weq{2}{
\frac{\vartheta'_{11}\vartheta_{10}(v+a)}{\pi\vartheta_{11}(v)\vartheta_{10}(a)}
= \cotg \pi v - \tg \pi a + 4\sum(-1)^{\frac{m}{2}} q^{\frac{mm'}{2}}\sin \pi (m a + m' v).
}

\weq{3}{
\frac{\vartheta'_{11}\vartheta_{01}(v+a)}{\pi\vartheta_{11}(v)\vartheta_{01}(a)}
= \frac{1}{\sin \pi v} + 4\sum q^{\frac{mn}{2}}\sin \pi (m a + n v).
}

\weq{4}{
\frac{\vartheta'_{11}\vartheta_{00}(v+a)}{\pi\vartheta_{11}(v)\vartheta_{00}(a)}
= \frac{1}{\sin \pi v} + 4\sum(-1)^{\frac{m}{2}} q^{\frac{mn}{2}}\sin \pi (m a + n v).
}

\weq{5}{
\frac{\vartheta'_{11}\vartheta_{10}(v+a)}{\pi\vartheta_{10}(v)\vartheta_{11}(a)}
\qquad \text{Vertauschung von } a \text{ und } v \text{ in (2).}
}

\weq{6}{
\begin{aligned}[t]
\frac{\vartheta'_{11}\vartheta_{11}(v+a)}{\pi\vartheta_{10}(v)\vartheta_{10}(a)} &= \tg \pi v + \tg \pi a \\
&\quad {} - 4\sum(-1)^{\frac{m+m'}{2}} q^{\frac{mm'}{2}}\sin \pi (m a + n' v).
% [sic] printed $n'$ in $\sin \pi (m a + n' v)$, should be $m'$: the summation letters of (6) are $m, m'$ (cf. $(-1)^{\frac{m+m'}{2}} q^{\frac{mm'}{2}}$), and (1) with $v \to v+\frac{1}{2}$, $a \to a-\frac{1}{2}$ gives $\sin \pi (m a + m' v)$
\end{aligned}
}

\weq{7}{
\begin{aligned}[t]
\frac{\vartheta'_{11}\vartheta_{00}(v+a)}{\pi\vartheta_{10}(v)\vartheta_{01}(a)} &= \frac{1}{\cos \pi v} \\
&\quad {} + 4\sum(-1)^{\frac{n-2}{2}} q^{\frac{mn}{2}}\cos \pi (m a + n v).
% [sic] printed exponent $\frac{n-2}{2}$, should be $\frac{n-1}{2}$: $n$ is odd, so $\frac{n-2}{2}$ is not an integer; (3) with $v \to v+\frac{1}{2}$ gives $(-1)^{\frac{n-1}{2}}$, consistent with $\frac{m+n-1}{2}$ in (8)
\end{aligned}
}

\weq{8}{
\frac{\vartheta'_{11}\vartheta_{01}(v+a)}{\pi\vartheta_{10}(v)\vartheta_{00}(a)}
= \frac{1}{\cos \pi v} + 4\sum(-1)^{\frac{m+n-1}{2}} q^{\frac{mn}{2}}\cos \pi (m a + n v).
}
